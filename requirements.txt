# AstroGuide - Dependências do Projeto

# pip install -r requirements.txt - Para instalar todas as dependências

# Servidor web – recebe pedidos do browser e devolve respostas
flask==3.1.1
 
# Motor astronómico – calcula posições de planetas, lua, sol
# usa as efemérides da NASA (de421.bsp)

# Base de dados de fusos horários – necessário para skyfield
# calcular horas locais corretamente em qualquer país
tzdata==2025.2

# Pedidos HTTP – usado para ir buscar a Imagem Astronómica do Dia (APOD) à NASA
requests==2.32.3

# AstroGuide AI – a API Groq é chamada com o requests (em cima); não há SDK
# nenhum para ela. O modelo e a chave vêm das variáveis de ambiente
# GROQ_MODEL e GROQ_API_KEY (ver .env.example).

# AstroGuide AI (alternativa Gemini) – o SDK do Google Gemini, usado pelo
# py/ia/ai_engine.py quando AI_PROVIDER=gemini. É o único pacote do projeto
# que serve uma funcionalidade opcional: sem ele a aplicação corre igual, e é
# só o caminho Gemini — nunca o principal — que fica indisponível.
google-genai==2.28.0

# Variáveis de ambiente – carrega o ficheiro .env para o ambiente no arranque
# do server.py, para correr na própria máquina sem exportar variáveis à mão.
# Também opcional: sem ele, as variáveis vêm do ambiente de sempre.
python-dotenv==1.0.1
